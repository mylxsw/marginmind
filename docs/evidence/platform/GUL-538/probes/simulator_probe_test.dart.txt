// GUL-538 scratch probe (not committed): drive the real app on a simulator,
// print SHOT markers for external `simctl io screenshot`, and collect layout
// overflow errors.
import 'dart:io';

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:marginmind/main.dart' as app;

Future<void> idle(WidgetTester t, int seconds) async {
  for (var i = 0; i < seconds * 10; i++) {
    await t.pump(const Duration(milliseconds: 100));
  }
}

/// The marker file is only reachable on simulators (shared host /tmp);
/// Android runs read the SHOT lines from the test output instead.
void _mark(String name) {
  try {
    File('/tmp/gul538/marker').writeAsStringSync(name);
  } on FileSystemException {
    // Not on a simulator.
  }
}

Future<void> shot(WidgetTester t, String name) async {
  // ignore: avoid_print
  print('SHOT:$name');
  _mark(name);
  await idle(t, 6);
}

void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  testWidgets('probe', (t) async {
    final errors = <String>[];
    final previous = FlutterError.onError;
    FlutterError.onError = (d) => errors.add(d.exceptionAsString().split('\n').first);
    app.main();
    await idle(t, 6);
    await shot(t, 'library');
    final sample = find.text('Open Sample Study Set');
    if (sample.evaluate().isNotEmpty) {
      await t.tap(sample.first, warnIfMissed: false);
    } else {
      final card = find.textContaining('MarginMind');
      if (card.evaluate().isNotEmpty) await t.tap(card.first, warnIfMissed: false);
    }
    await idle(t, 8);
    await shot(t, 'workspace');
    final size = t.view.physicalSize / t.view.devicePixelRatio;
    final pad = MediaQueryData.fromView(t.view).padding;
    // ignore: avoid_print
    print('PROBE logical=$size safeArea=$pad');
    await SystemChrome.setPreferredOrientations([DeviceOrientation.landscapeLeft]);
    await idle(t, 4);
    await shot(t, 'workspace-landscape');
    // ignore: avoid_print
    print('PROBE landscape logical=${t.view.physicalSize / t.view.devicePixelRatio}');
    await SystemChrome.setPreferredOrientations([]);
    FlutterError.onError = previous;
    final counts = <String, int>{};
    for (final e in errors) {
      counts[e] = (counts[e] ?? 0) + 1;
    }
    // ignore: avoid_print
    counts.forEach((k, v) => print('PROBE error x$v: $k'));
    _mark('done');
  });
}
