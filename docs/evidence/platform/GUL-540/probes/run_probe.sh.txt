#!/bin/bash
# GUL-540 scratch run on the Mac worker. usage: run_probe.sh setup | run <device name> <label>
set -u
export PATH=$HOME/fvm/default/bin:/opt/homebrew/bin:$PATH
W=~/macrun-work/mm-gul540
BRANCH=agent/arch-opus-ora/6db9c4422c51
case "$1" in
setup)
  if [ ! -d $W ]; then git clone -q https://github.com/mylxsw/marginmind.git $W || exit 1; fi
  cd $W && git fetch -q origin $BRANCH && git checkout -q -B gul540 FETCH_HEAD && git log --oneline -1
  cd apps/marginmind
  [ -d ios ] || cp -R ~/macrun-work/mm-gul538/apps/marginmind/ios ios
  rm -rf ios/Pods ios/Podfile.lock ios/Flutter/ephemeral
  # Scratch only: a full-screen app lets the probe rotate the iPad.
  /usr/libexec/PlistBuddy -c "Delete :UIRequiresFullScreen" ios/Runner/Info.plist 2>/dev/null
  /usr/libexec/PlistBuddy -c "Add :UIRequiresFullScreen bool true" ios/Runner/Info.plist
  mkdir -p integration_test && cp ~/macrun-work/gul540-probe/probe_test.dart integration_test/probe_test.dart
  grep -q 'integration_test:' pubspec.yaml || sed -i '' 's/^dev_dependencies:/dev_dependencies:\
  integration_test:\
    sdk: flutter/' pubspec.yaml
  flutter pub get 2>&1 | tail -1
  ;;
run)
  NAME=$2; LABEL=$3
  UDID=$(xcrun simctl list devices available | grep -F "$NAME (" | head -1 | grep -oE '[0-9A-F-]{36}')
  echo "UDID=$UDID"
  cd $W/apps/marginmind
  mkdir -p /tmp/gul540 ~/macrun-work/gul540-shots; rm -f /tmp/gul540/marker
  xcrun simctl boot $UDID 2>/dev/null; xcrun simctl bootstatus $UDID -b >/dev/null
  xcrun simctl uninstall $UDID dev.marginmind.marginmind 2>/dev/null
  ( last=""; for i in $(seq 1 1800); do m=$(cat /tmp/gul540/marker 2>/dev/null); if [ -n "$m" ] && [ "$m" != "$last" ]; then [ "$m" = done ] && break; sleep 2.5; xcrun simctl io $UDID screenshot ~/macrun-work/gul540-shots/$LABEL-$m.png >/dev/null 2>&1; last=$m; fi; sleep 0.5; done ) &
  P=$!
  flutter test integration_test/probe_test.dart -d $UDID 2>&1 | grep -E "PROBE|SHOT|passed|failed|Error|rror:" | head -80
  wait $P; ls ~/macrun-work/gul540-shots | grep "^$LABEL"
  ;;
esac
